# =============================================================
# requirements.txt — Python dependencies
#
# Install everything with: pip install -r requirements.txt
#
# System dependencies (not installed by pip):
#   FFmpeg      — video rendering; install with: brew install ffmpeg
#   whisper.cpp — free local speech-to-text for the clips channel;
#                 install with: brew install whisper-cpp
#                 (chosen over the openai-whisper / faster-whisper Python
#                 packages because their libraries don't support Python
#                 3.15 yet; whisper.cpp is a separate program like FFmpeg)

# Anthropic SDK — talks to the Claude API

# RSS feed parser — reads news feeds from NPR, BBC, etc.
feedparser>=6.0.0

# YAML parser — reads channel config files (channels/*.yaml)
PyYAML>=6.0.0

# python-dotenv — loads .env into environment variables at startup
python-dotenv>=1.0.0

# edge-tts — free neural text-to-speech via Microsoft Edge's engine
#             no API key needed; provides word-level timing for captions
edge-tts>=6.1.0

# Pillow — image generation for video frames (text, background)
Pillow>=10.0.0

# pydantic — data validation; standard in FastAPI pipelines
pydantic>=2.0.0

# FastAPI — web framework for the dashboard (industry standard, used at Meta, Netflix, etc.)
fastapi>=0.110.0

# uvicorn — ASGI server that runs FastAPI
uvicorn>=0.29.0

# Jinja2 — HTML templates for the dashboard pages; the standard template
#          engine for FastAPI. Escapes text automatically (blocks HTML injection).
Jinja2>=3.1.0

# bcrypt — hashes the dashboard password; the most widely used
#          password-hashing library (deliberately slow, so guessing is impractical)
bcrypt>=4.0.0

# itsdangerous — signs the login cookie; required by Starlette's SessionMiddleware
itsdangerous>=2.1.0

# PyCA cryptography — authenticated encryption for per-channel account tokens.
# Compared with PyCryptodome by GitHub adoption; Fernet supplies a small safe API.

# pytest — runs the test suite
pytest>=8.0.0

# httpx — async HTTP client used by FastAPI's test client
httpx>=0.27.0

# python-multipart — required by FastAPI for form data (approve/reject buttons)
python-multipart>=0.0.9

# numpy — fast number arrays; used to reshape voiceover pauses (src/video/pacing.py).
#         Already installed with matplotlib; listed because we import it directly.
numpy>=1.26.0

# matplotlib — local chart generation for visuals (Phase V.5); no API key needed
matplotlib>=3.8.0

# lottie — Lottie animation format parser (kept for future rendering once
#           cairo/pango are available via brew install cairo pango).
#           Currently unused at runtime — Pillow animations are used instead.
lottie>=0.7.2

# faster-whisper — Whisper model via CTranslate2; used for word-level timing
#                  when tts_provider is "macos" or "kokoro", and selectable for
#                  stream clips (transcription.engine: faster-whisper; whisper.cpp
#                  is preferred there because it kept shouted speech in testing).
#                  PyAV now ships a Python 3.15-compatible wheel (av 19).
#   pip install: pip install --no-deps faster-whisper tokenizers huggingface_hub
#   pip install: pip install av==19.0.2   (faster-whisper imports it)
ctranslate2>=4.0.0
tokenizers>=0.13.4
huggingface_hub>=0.22.0
# Voice recognition so captions show only the streamer's words (src/clips/speaker_filter.py).
# Chosen over pyannote/SpeechBrain: no PyTorch, one package, supports Python 3.15.
sherpa-onnx==2.0.2
# OpenCV (headless) — face detection for face-aware clip framing (src/clips/framing.py)
#   using the YuNet model (MIT, 230 KB, auto-downloaded to models/face/ and hash-checked).
#   The most-used computer-vision library (Apache-2.0); "headless" = no GUI parts.
#   Chosen over MediaPipe (larger, Google-specific runtime) and dlib (needs compiling).
opencv-python-headless==5.1.0.95
